
--- Model_Selected_Final-96,1.ipynb ---
MODEL/COMPILE CELL:
num_classes = y.shape[1]
model = Sequential([
    Masking(mask_value=0.0, input_shape=(30, 696)),
    LayerNormalization(),

    # LSTM 1 — giữ 128 nhưng giảm recurrent_dropout
    LSTM(128, return_sequences=True, activation='tanh', recurrent_dropout=0.1),
    BatchNormalization(),
    Dropout(0.4),

    # LSTM 2 — giảm từ 128 → 64 (bớt params)
    LSTM(64, return_sequences=False, activation='tanh', recurrent_dropout=0.1),
    BatchNormalization(),
    Dropout(0.4),

    # Dense 1 — giảm từ 256 → 128
    Dense(128, activation='relu', kernel_regularizer=regularizers.l2(0.002)),
    Dropout(0.3),

    # Bỏ Dense 128 thứ 2 — không cần thiết với data nhỏ
    Dense(num_classes, activation='softmax')
])
MODEL/COMPILE CELL:
initial_lr   = 0.001
decay_steps  = 5000
lr_schedule  = tf.keras.optimizers.schedules.CosineDecayRestarts(
    initial_learning_rate=initial_lr,
    first_decay_steps=decay_steps,
    t_mul=2.0, m_mul=0.9, alpha=1e-5
)

optimizer = Adam(learning_rate=lr_schedule)
loss      = tf.keras.losses.CategoricalCrossentropy(label_smoothing=0.05)  # 0.1→0.05

model.compile(optimizer=optimizer, loss=loss, metrics=['categorical_accuracy'])
model.summary()
